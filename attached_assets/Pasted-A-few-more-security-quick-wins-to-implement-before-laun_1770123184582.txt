async getWheelCountByUserId(userId: number): Promise<number> {
  const result = await db
    .select({ count: sql<number>`count(*)` })
    .from(wheels)
    .where(eq(wheels.userId, userId));
  return Number(result[0]?.count ?? 0);
}

QUICK FIX 5: BASIC CSRF PROTECTION

For session-based auth, add a simple CSRF check. The easiest approach:

1. Check that state-changing requests (POST, PUT, DELETE) have the correct Content-Type header
2. Browsers won't send application/json from a form submission attack

Add this middleware before your routes:

const csrfProtection = (req: Request, res: Response, next: NextFunction) => {
  if (["POST", "PUT", "DELETE", "PATCH"].includes(req.method)) {
    const contentType = req.headers["content-type"];
    if (!contentType || !contentType.includes("application/json")) {
      return res.status(403).json({ error: "Invalid content type" });
    }
  }
  next();
};

app.use("/api", csrfProtection);

This is a simple protection — forms can't send application/json without JavaScript.
